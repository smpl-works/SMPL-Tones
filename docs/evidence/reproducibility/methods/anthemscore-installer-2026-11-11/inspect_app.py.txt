"""Inspect native imports and literal references without loading AnthemScore."""
from concurrent.futures import ThreadPoolExecutor
from pathlib import Path
import hashlib
import json
import re
import struct
import subprocess

ROOT = Path(__file__).resolve().parent
PATH = ROOT/'payload/core/AnthemScore.exe'
data = PATH.read_bytes()
NODE = r'<NODE_EXECUTABLE>'
REA = r'<REA_CLI_ENTRY>'


def inspect(command):
    r=subprocess.run([NODE,REA,command,str(PATH),'--format','json','--full-output'],capture_output=True,text=True,encoding='utf-8',timeout=60)
    (ROOT/f'app-{command}.json').write_text(r.stdout,encoding='utf-8')
    d=json.loads(r.stdout);e=d.get('data',{});n=e.get('normalized_result',{})
    return {'command':command,'exit':r.returncode,'evidence_id':e.get('evidence_id'),
            'classification':n.get('classification'),'coverage':n.get('coverage'),'error':d.get('error')}


def cstring(offset):
    return data[offset:data.index(b'\x00',offset)].decode('ascii',errors='replace')


def inspect_bytes():
    pe=struct.unpack_from('<I',data,60)[0]
    machine,count=struct.unpack_from('<HH',data,pe+4)
    optional_size=struct.unpack_from('<H',data,pe+20)[0]
    optional=pe+24
    magic=struct.unpack_from('<H',data,optional)[0]
    directory=optional+(112 if magic==0x20b else 96)
    sections=[]
    for i in range(count):
        pos=optional+optional_size+40*i
        name=data[pos:pos+8].rstrip(b'\x00').decode()
        virtual_size,rva,size,offset=struct.unpack_from('<IIII',data,pos+8)
        sections.append({'name':name,'rva':rva,'virtual_bytes':virtual_size,'file_bytes':size,'offset':offset})
    def file_offset(rva):
        for section in sections:
            if section['rva']<=rva<section['rva']+max(section['virtual_bytes'],section['file_bytes']):
                return section['offset']+rva-section['rva']
        raise ValueError(hex(rva))
    imports=[]
    start=file_offset(struct.unpack_from('<I',data,directory+8)[0])
    for i in range(256):
        original,timestamp,forwarder,name,first=struct.unpack_from('<IIIII',data,start+20*i)
        if not any([original,timestamp,forwarder,name,first]):break
        thunk=file_offset(original or first)
        width=8 if magic==0x20b else 4
        functions=[]
        for j in range(65536):
            entry=int.from_bytes(data[thunk+width*j:thunk+width*(j+1)],'little')
            if not entry:break
            functions.append({'ordinal':entry&0xffff} if entry&(1<<(width*8-1)) else {'name':cstring(file_offset(entry)+2)})
        imports.append({'dll':cstring(file_offset(name)),'functions':functions})
    (ROOT/'app-imports.json').write_text(json.dumps({'sha256':hashlib.sha256(data).hexdigest(),'machine':hex(machine),'sections':sections,'imports':imports},indent=2))
    pattern=re.compile(r'onnx|tensorflow|tflite|\.onnx|spectro|fourier|fft|neural|predict|quantiz|\.asdt|\.xml|\.musicxml|\.midi|printToPdf|OpenSheetMusic|AS_resources|models[/\\]|\.pb|\.h5|instrument.?detect|beat.?detect',re.I)
    literals=[]
    for encoding,regex in [('ascii',rb'[\x20-\x7e]{7,}'),('utf-16le',rb'(?:[\x20-\x7e]\x00){7,}'),('utf-16be',rb'(?:\x00[\x20-\x7e]){7,}')]:
        for match in re.finditer(regex,data):
            text=match.group().decode(encoding)
            if pattern.search(text):literals.append({'offset':match.start(),'encoding':encoding,'text':text})
    (ROOT/'app-selected-literals.json').write_text(json.dumps(literals,indent=2),encoding='utf-8')
    focused=[x for x in literals if re.search(r'\.onnx|\.asdt|OpenSheetMusic|AS_resources|neural|printToPdf|model[s/\\]|predict|spectro',x['text'],re.I)]
    return {'sha256':hashlib.sha256(data).hexdigest(),'bytes':len(data),'literal_count':len(literals),
            'imports':[{'dll':x['dll'],'functions':len(x['functions'])} for x in imports],
            'focused_literals':focused[:80]}


if __name__=='__main__':
    with ThreadPoolExecutor(max_workers=2) as pool:
        results=list(pool.map(inspect,['inspect-managed-artifact','inspect-pe-resources']))
    print(json.dumps({'rea':results,'bytes':inspect_bytes()},indent=2))
