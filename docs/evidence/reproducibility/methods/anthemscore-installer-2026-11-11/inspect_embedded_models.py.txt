"""Read embedded ONNX graph metadata without extracting weights or running inference.

Field identities follow onnx/onnx.proto; source reference is retained separately.
"""
from collections import Counter
from pathlib import Path
import hashlib
import json
import re
import struct

ROOT=Path(__file__).resolve().parent
data=(ROOT/'payload/core/AnthemScore.exe').read_bytes()


def varint(buf,pos):
    value=0
    for shift in range(0,70,7):
        byte=buf[pos];pos+=1;value|=(byte&127)<<shift
        if byte<128:return value,pos
    raise ValueError('Invalid varint')


def fields(buf):
    pos=0
    while pos<len(buf):
        tag,pos=varint(buf,pos)
        number,wire=tag>>3,tag&7
        assert number>0
        if wire==0:value,pos=varint(buf,pos)
        elif wire==2:
            length,pos=varint(buf,pos)
            assert pos+length<=len(buf)
            value=buf[pos:pos+length];pos+=length
        elif wire in (1,5):
            length=8 if wire==1 else 4
            assert pos+length<=len(buf)
            value=buf[pos:pos+length];pos+=length
        else:raise ValueError(f'Unsupported wire type {wire}')
        yield number,wire,value


def text(value):return bytes(value).decode('utf-8')


def value_info(buf):
    result={}
    for number,wire,value in fields(buf):
        if number==1:result['name']=text(value)
        elif number==2:
            for n,w,v in fields(value):
                if n!=1:continue
                for tn,tw,tv in fields(v):
                    if tn==1:result['element_type_id']=tv
                    elif tn==2:
                        dims=[]
                        for sn,sw,sv in fields(tv):
                            if sn!=1:continue
                            dim=None
                            for dn,dw,dv in fields(sv):
                                if dn==1:dim=dv
                                elif dn==2:dim=text(dv)
                            dims.append(dim)
                        result['shape']=dims
    return result


models=[]
for candidate in re.finditer(rb'\x08\x08\x12\x07pytorch',data):
    start=candidate.start()
    length=struct.unpack_from('>I',data,start-4)[0]
    assert 0<length<64*1024*1024 and start+length<=len(data)
    model=memoryview(data)[start:start+length]
    item={'offset':start,'length_prefix_offset':start-4,'bytes':length,
          'sha256':hashlib.sha256(model).hexdigest()}
    graph=None
    for number,wire,value in fields(model):
        if number==1:item['ir_version']=value
        elif number==2:item['producer']=text(value)
        elif number==3:item['producer_version']=text(value)
        elif number==7:graph=value
        elif number==8:
            opset={}
            for n,w,v in fields(value):
                if n==1:opset['domain']=text(v)
                elif n==2:opset['version']=v
            item.setdefault('opsets',[]).append(opset)
        elif number==14:
            pair={n:text(v) for n,w,v in fields(value)}
            item.setdefault('metadata',{})[pair.get(1,'')]=pair.get(2,'')
    assert graph is not None
    operators=Counter();initializers=0;initializer_types=Counter()
    for number,wire,value in fields(graph):
        if number==1:
            name=next((text(v) for n,w,v in fields(value) if n==4),'unknown')
            operators[name]+=1
        elif number==2:item['graph_name']=text(value)
        elif number==5:
            initializers+=1
            datatype=next((v for n,w,v in fields(value) if n==2),None)
            initializer_types[datatype]+=1
        elif number==11:item.setdefault('inputs',[]).append(value_info(value))
        elif number==12:item.setdefault('outputs',[]).append(value_info(value))
    item.update({'nodes':sum(operators.values()),'operators':dict(operators),
                 'initializers':initializers,'initializer_element_types':dict(initializer_types)})
    models.append(item)
assert models
result={'parent_sha256':hashlib.sha256(data).hexdigest(),'method':'Length-prefixed ONNX ModelProto metadata decoding; no inference or weight export',
        'model_count':len(models),'unique_model_sha256_count':len({x['sha256'] for x in models}),'models':models}
(ROOT/'embedded-models.json').write_text(json.dumps(result,indent=2),encoding='utf-8')
print(json.dumps({'models':[{k:v for k,v in x.items() if k not in ['operators','initializers','initializer_element_types','sha256']} for x in models],
                  'unique_model_count':result['unique_model_sha256_count']},indent=2))
